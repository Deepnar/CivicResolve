## 🚀 **Getting Started with AI Features**

### **1. Environment Setup for AI**

```bash
# Clone the repository
git clone https://github.com/yourusername/civicresolve.git
cd civicresolve

# Install dependencies (includes AI packages)
npm install
# or
pnpm install

# Set up environment variables (including AI configuration)
cp .env.example .env.local
```

### **2. AI Configuration**

Add your Google Gemini AI API key to `.env.local`:

```bash
# Google Gemini AI Configuration
GEMINI_API_KEY=your_google_ai_api_key_here

# Database Configuration
DB_HOST=localhost
DB_PORT=3306
DB_USER=your_db_user
DB_PASSWORD=your_db_password
DB_NAME=civicresolve

# JWT Secret for AI-aware authentication
JWT_SECRET=your_super_secret_jwt_key_here
```

### **3. Database Setup with AI Schema**

```bash
# Create database with AI-optimized schema
mysql -u root -p < database-schema.sql

# Run initialization scripts
mysql -u root -p civicresolve < scripts/init-database.sql
mysql -u root -p civicresolve < scripts/seed-data.sql
```

### **4. Launch the AI-Powered Platform**

```bash
# Development mode with AI features
npm run dev
# or
pnpm run dev

# Production build with AI optimization
npm run build
npm start
```

The platform will be available at `http://localhost:3000` with full AI capabilities.

## 🔧 **Technical AI Implementation**

### **🤖 Google Gemini Integration**

```typescript
// AI Configuration
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);
const model = genAI.getGenerativeModel({ model: "gemini-2.0-flash-exp" });

// Real-time data integration
async function generateAIResponse(userMessage: string, context: UserContext) {
  const platformStats = await getPlatformStatistics();
  const userStats = await getCurrentUserStats(context.userId);
  
  const systemPrompt = buildIntelligentPrompt({
    userRole: context.role,
    platformData: platformStats,
    personalData: userStats,
    currentPage: context.page,
    securityLevel: determineSecurityLevel(context.role)
  });

  return await model.generateContent([
    { text: systemPrompt },
    { text: userMessage }
  ]);
}
```

### **🧠 Intelligent Data Processing**

```typescript
// Real-time platform statistics for AI analysis
async function getPlatformStatistics() {
  return {
    totalIssues: await Database.query('SELECT COUNT(*) FROM issues'),
    issuesByCategory: await Database.query(`
      SELECT category, COUNT(*) as count 
      FROM issues GROUP BY category ORDER BY count DESC
    `),
    votingTrends: await Database.query(`
      SELECT i.id, i.title, COUNT(v.id) as votes
      FROM issues i LEFT JOIN votes v ON i.id = v.issue_id
      GROUP BY i.id ORDER BY votes DESC LIMIT 10
    `),
    resolutionMetrics: await Database.query(`
      SELECT AVG(DATEDIFF(updated_at, created_at)) as avg_days,
      category FROM issues WHERE status = 'RESOLVED'
      GROUP BY category
    `)
  };
}
```

```sql
CREATE TABLE users (
```

```sql
CREATE TABLE issues (
  id VARCHAR(255) PRIMARY KEY,
  title VARCHAR(255) NOT NULL,
  description TEXT NOT NULL,
  category ENUM('ROADS', 'UTILITIES', 'SAFETY', 'ENVIRONMENT', 'OTHER') NOT NULL,
  priority ENUM('LOW', 'MEDIUM', 'HIGH', 'URGENT') DEFAULT 'MEDIUM',
  status ENUM('PENDING', 'IN_PROGRESS', 'RESOLVED') DEFAULT 'PENDING',
  location_lat DECIMAL(10, 8) NOT NULL,
  location_lng DECIMAL(11, 8) NOT NULL,
  address TEXT,
  images JSON,
  reporter_id VARCHAR(255) NOT NULL,
  reporter_name VARCHAR(255) NOT NULL,
  votes_count INT DEFAULT 0,
  comments_count INT DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (reporter_id) REFERENCES users(id) ON DELETE CASCADE
);
```

#### **Comments Table**
```sql
CREATE TABLE comments (
  id VARCHAR(255) PRIMARY KEY,
  issue_id VARCHAR(255) NOT NULL,
  author_id VARCHAR(255) NOT NULL,
  author_name VARCHAR(255) NOT NULL,
  content TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (issue_id) REFERENCES issues(id) ON DELETE CASCADE,
  FOREIGN KEY (author_id) REFERENCES users(id) ON DELETE CASCADE
);
```

#### **Votes Table**
```sql
CREATE TABLE votes (
  id VARCHAR(255) PRIMARY KEY,
```

### **Authentication & Security**

#### **JWT Implementation**
```typescript
// Token generation
const token = jwt.sign(
  { userId: user.id, email: user.email, role: user.role },
  process.env.JWT_SECRET!,
  { expiresIn: '7d' }
);

// Cookie configuration
response.cookies.set('auth-token', token, {
  httpOnly: true,
  secure: process.env.NODE_ENV === 'production',
  sameSite: 'strict',
  maxAge: 7 * 24 * 60 * 60 * 1000 // 7 days
});
```

#### **Middleware Protection**
```typescript
// Route protection middleware
export async function middleware(request: NextRequest) {
  const token = request.cookies.get('auth-token')?.value;
  
  if (request.nextUrl.pathname.startsWith('/admin')) {
    // Admin routes require authentication and ADMIN role
    const user = await verifyToken(token);
    if (!user || user.role !== 'ADMIN') {
      return NextResponse.redirect(new URL('/login', request.url));
    }
  }
  
  if (request.nextUrl.pathname.startsWith('/profile')) {
    // Profile routes require authentication
    const user = await verifyToken(token);
    if (!user) {
      return NextResponse.redirect(new URL('/login', request.url));
    }
  }
}
```

```typescript
const handleVoteOptimistic = async (issueId: string) => {
  // Immediately update UI
  setVoteCount(prev => hasVoted ? prev - 1 : prev + 1);
  setHasVoted(!hasVoted);
  
  try {
    // Make API call
    const response = await fetch(`/api/issues/${issueId}/vote`, {
```
